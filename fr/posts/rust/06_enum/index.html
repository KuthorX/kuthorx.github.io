  • Syntaxe de base
   // déclaration de base
   enum IpAddrKind {
       V4,
       V6,
   }
   // utilisation de base
   let four = IpAddrKind::V4;
   let six = IpAddrKind::V6;
   
   // utilisable dans un struct
   struct IpAddr {
       kind: IpAddrKind,
       address: String,
   }
   
   // peut contenir différents type, y compris des struct
   enum IpAddr {
       V4(u8, u8, u8, u8),
       V6(String),
   }
   let home = IpAddr::V4(127, 0, 0, 1);
   let loopback = IpAddr::V6(String::from("::1"));
   struct Ipv4Addr {
       // --snip--
   }
   struct Ipv6Addr {
       // --snip--
   }
   enum IpAddr {
       V4(Ipv4Addr),
       V6(Ipv6Addr),
   }
   enum Message {
       Quit,
       Move { x: i32, y: i32 },
       Write(String),
       ChangeColor(i32, i32, i32),
   }
   
   // comme pour un struct, impl est supporté
   impl Message {
       fn call(&self) {
           // method body would be defined here
       }
   }

    match,

   #[derive(Debug)]
   enum UsState {
       Alabama,
       Alaska,
       // --snip--
   }
   #[derive(Debug)]
   enum Coin {
       Penny,
       Nickel,
       Dime,
       Quarter(UsState),
   }
   fn value_in_cents(coin: Coin) -> u8 {
       match coin {
           Coin::Penny => 1,  // quand il n'y a qu'une ligne, on peut omettre {}
           Coin::Nickel => 5,
           Coin::Dime => 10,
           Coin::Quarter(state) => {
   						// cette valeur peut être UsState::Alabama ou UsState::Alaska
               println!("State quarter from {:?}!", state);
               25
           }
       };
       match coin {
           Coin::Penny => 1,
   				// on peut utiliser une autre variable pour les autres cas
           a => {
               println!("{:?}", a);
               5
           },
       };
       match coin {
           Coin::Penny => 1,
   				// on peut utiliser un underscore
           _ => 5,
       }
   }
  • if let
   let mut count = 0;
   
   // écriture avec match
   match coin {
       Coin::Quarter(state) => println!("State quarter from {:?}!", state),
       _ => count += 1,
   }
   
   // écriture avec if let
   if let Coin::Quarter(state) = coin {
       println!("State quarter from {:?}!", state);
   } else {
       count += 1;
   }

    x: T,
    y: T,

impl<T> Point<T> {
    fn x(&self) -> &T {
        &self.x
    }
}

// 泛型声明 - enum

    Some(T),
    None,
}

    Ok(T),
    Err(E),

// 一种 mix-up 实现
struct Point<T, U> {
    x: T,
    y: U,
}
impl<T, U> Point<T, U> {
		// 由于泛型的高灵活性，这里做了巧妙的 field 交叉混合，组合出第三个新泛型实例
    // 这种语法可用于 middleware
    fn mixup<V, W>(self, other: Point<V, W>) -> Point<T, W> {
        Point {
            x: self.x,
            y: other.y,
        }
    }
}
fn main() {
    let p1 = Point { x: 5, y: 10.4 };
    let p2 = Point { x: "Hello", y: 'c' };
    let p3 = p1.mixup(p2);
    println!("p3.x = {}, p3.y = {}", p3.x, p3.y);
}

    컴파일러는 명시적 주석이 없을 때 reference의 lifetime을 계산하기 위해 세 가지 규칙을 사용합니다. 첫 번째 규칙은 reference인 각 매개변수가 고유한 lifetime 매개변수를 갖는다는 것입니다. 첫 번째: 각 reference 입력 매개변수에는 자체 lifetime 매개변수가 있습니다. 두 번째 규칙은 입력 lifetime 매개변수가 정확히 하나라면 그 lifetime을 모든 출력 lifetime 매개변수에 할당한다는 것입니다. 두 번째: 입력 lifetime 매개변수가 하나뿐이면 이 매개변수를 모든 출력 lifetime 매개변수에 할당합니다. 세 번째 규칙은 입력 lifetime 매개변수가 여러 개이고 그중 하나가 method의 &self 또는 &mut self라면 self의 lifetime을 모든 출력 lifetime 매개변수에 할당한다는 것입니다. 세 번째: 입력 lifetime 매개변수가 여러 개이고 그중 하나가 &self 또는 &mut self(method)라면 self의 lifetime 매개변수를 모든 출력 lifetime 매개변수에 할당합니다.

제네릭을 사용하는 코드의 성능

generic type 매개변수를 사용할 때 런타임 비용이 있는지 궁금할 수 있습니다. 좋은 소식은 Rust가 제네릭을 구현하는 방식 덕분에 generic type을 사용해도 concrete type을 사용할 때보다 코드가 느려지지 않는다는 것입니다. Rust는 컴파일 시 제네릭을 사용하는 코드를 monomorphization하여 이를 달성합니다. Monomorphization은 컴파일할 때 사용되는 concrete type을 채워 generic code를 구체적인 코드로 바꾸는 과정입니다. … 컴파일러는 명시적 주석이 없을 때 reference의 lifetime을 알아내기 위해 세 규칙을 사용합니다. 첫 번째 규칙은 입력 lifetime에, 두 번째와 세 번째 규칙은 출력 lifetime에 적용됩니다. 세 규칙을 모두 적용한 뒤에도 lifetime을 알아낼 수 없는 reference가 남으면 컴파일러는 오류와 함께 중단합니다. 이 규칙은 fn 정의와 impl 블록 모두에 적용됩니다. 첫 번째 규칙은 reference인 각 매개변수가 고유한 lifetime 매개변수를 갖는다는 것입니다. In other words, a function with one parameter gets one lifetime parameter: fn foo<'a>(x: &'a i32); a function with two parameters gets two separate lifetime parameters: fn foo<'a, 'b>(x: &'a i32, y: &'b i32); and so on. 두 번째 규칙은 입력 lifetime 매개변수가 정확히 하나라면 그 lifetime을 모든 출력 lifetime 매개변수에 할당한다는 것입니다.: fn foo<'a>(x: &'a i32) -> &'a i32. 세 번째 규칙은 입력 lifetime 매개변수가 여러 개이고 그중 하나가 method의 &self 또는 &mut self라면 self의 lifetime을 모든 출력 lifetime 매개변수에 할당한다는 것입니다.This third rule makes methods much nicer to read and write because fewer symbols are necessary.
